58 //! @addtogroup quaternion
59 //! @{
60 //! @var from_euler
61 //!
62 //! @brief Callable object computing a quaternion from its euler representation.
63 //!
64 //! This function builds a quaternion from 3 euler angles in radian
65 //! (or in \f$\pi multiples\f$ if the option `radpi` is used).
66 //! Template parameters I, J, K of type int
67 //! are used to choose the euler axis order.
68 //!
69 //! for instance I = 3, J = 2, K = 3 choose the ZYZ sequence.
70 //! the values of I, J, and K must be in {1, 2, 3} ans satisfy I != J && J != K.
71 //!
72 //! @groupheader{Header file}
73 //!
74 //! @code
75 //! #include kyosu/quaternion.hpp>`
76 //! @endcode
77 //!
78 //! @groupheader{Callable Signatures}
79 //!
80 //! @code
81 //! namespace eve
82 //! {
83 //! template < int I, int J, int K > auto from_euler(auto a, auto b, auto c
84 //! , axis<I> a1, axis<J> a2, axis<K> a3
85 //! ) const noexcept;
86 //! requires(I != J && J != K) //1
87 //!
88 //! template < int I, int J, int K > auto from_euler[extrinsic](auto a, auto b, auto c
89 //! , axis<I> a1, axis<J> a2, axis<K> a3
90 //! ) const noexcept;
91 //! requires(I != J && J != K) //1
92 //!
93 //! template < int I, int J, int K > auto from_euler[intrinsic](auto a, auto b, auto c
94 //! , axis<I> a1, axis<J> a2, axis<K> a3
95 //! ) const noexcept;
96 //! requires(I != J && J != K) //2
97 //! }
98 //! }
99 //! @endcode
100 //!
101 //! **Parameters**
102 //!
103 //! * `a`, `b`, `c`: the angles in radian
104 //! * `a1`, `a2`, `a3` the axis parameters to be chosen between X_, Y_, Z_ (two consecutive axis cannot be the same)
105 //!
106 //! **Template parameters**
107 //!
108 //! * I, J, K: are on call deduced from the axis parameters
109 //!
110 //! **Return value**
111 //!
112 //! 1. quaternion representing the rotation.
113 //! [extrinsic](https://en.wikipedia.org/wiki/Euler_angles) rotation scheme is used
114 //! 2. Same but the [intrinsic](https://en.wikipedia.org/wiki/Euler_angles) way is used
115 //! to interpret the angles.
